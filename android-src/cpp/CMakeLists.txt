cmake_minimum_required(VERSION 3.10)
project(sillonlame C)
file(GLOB LAME_SRC ${CMAKE_CURRENT_SOURCE_DIR}/lame/*.c)
add_library(sillonlame SHARED sillon_lame.c ${LAME_SRC})
target_include_directories(sillonlame PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lame)
target_compile_definitions(sillonlame PRIVATE HAVE_CONFIG_H)
target_compile_options(sillonlame PRIVATE -O2 -w)
target_link_libraries(sillonlame m)
# pages de 16 Ko (exigence des téléphones Android récents)
target_link_options(sillonlame PRIVATE "-Wl,-z,max-page-size=16384")
